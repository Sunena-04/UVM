module apb_slave (
    input logic pclk,
    input logic presetn,
    input logic psel,
    input logic penable,
    input logic pwrite,
    input logic [31:0] paddr,
    input logic [31:0] pwdata,
    output logic [31:0] prdata,
    output logic pready
);
    logic [31:0] mem [0:15];

    always_ff @(posedge pclk or negedge presetn) begin
        if (!presetn) begin
            prdata <= 0;
            pready <= 0;
        end else begin
            pready <= 0;
            if (psel && penable) begin
                pready <= 1;
                if (pwrite)
                    mem[paddr[5:2]] <= pwdata;
                else
                    prdata <= mem[paddr[5:2]];
            end
        end
    end
endmodule

interface apb_if(input logic pclk);

    logic presetn;
    logic psel;
    logic penable;
    logic pwrite;
    logic [31:0] paddr;
    logic [31:0] pwdata;
    logic [31:0] prdata;
    logic pready;

endinterface
